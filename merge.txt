DECLARE @UpperBound INT = 1000000;
DROP TABLE IF EXISTS ##Numbers
;WITH cteN(Number) AS
(
  SELECT ROW_NUMBER() OVER (ORDER BY s1.[object_id])
  FROM sys.all_columns AS s1
  CROSS JOIN sys.all_columns AS s2
)
SELECT [Number] INTO ##Numbers
FROM cteN WHERE [Number] <= @UpperBound;

DROP TABLE IF EXISTS ##SalesOrder
CREATE TABLE ##SalesOrder (Data_Ksiegowania date, Data_Sprzedazy date, Data_Wystawienia date, SalesAmount decimal(10,2));

DECLARE @StartDate AS date;
DECLARE @EndDate AS date;
DECLARE @DaysBetween AS int;
DECLARE @Count int;

SELECT @StartDate   = '01/01/2019',
       @EndDate     = '12/31/2021',
       @DaysBetween = (1+DATEDIFF(DAY, @StartDate, @EndDate)),
       @Count = 0;

WHILE (@Count < 10000000)
BEGIN
    
    ;WITH CTE AS (
	SELECT 
			DATEADD(DAY, RAND(CHECKSUM(NEWID()))*@DaysBetween,@StartDate) AS [Data_Sprzedazy]
			,ABS(CHECKSUM(NEWID()) % 50) + 10 AS [SalesAmount]
    FROM ##Numbers
	)
	INSERT INTO ##SalesOrder (Data_Sprzedazy,Data_Wystawienia, Data_Ksiegowania, SalesAmount)
	SELECT
			[Data_Sprzedazy]
			,DATEADD(DAY, 10,[Data_Sprzedazy]) AS 'Data_Wystawienia'
			,DATEADD(DAY, 20,[Data_Sprzedazy]) AS 'Data_Ksiegowania'
			,[SalesAmount]
	FROM CTE
   SET @Count = @Count + @@ROWCOUNT;
END

Odpowiedź:
Pierwsze polecenie, a więc MERGE z próbą zwrócenia output zwróci błąd z komunikatem „A nested INSERT, UPDATE, DELETE, or MERGE statement is not allowed in a SELECT statement that is not the immediate source of rows for an INSERT statement”, ponieważ klauzula OUTPUT została użyta w nieprawidłowy sposób. 

Drugie polecenie “Delete from dbo.DIM_TETA_USERS_SCD where [START_DATE] > [END_DATE]”
usunie wiersze z SCD gdzie data startu jest większa od daty końca obowiązywania wiersza. Ale tego typu sytuacja w SCD nie powinna się zdarzyć.
